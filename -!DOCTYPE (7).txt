/* HELPERS */
const $ = id => document.getElementById(id);
const LETTERS = ["A","B","C","D","E"];
const CACHE_PREFIX = "kpss_cache_v4_";
const CACHE_TTL = 1000 * 60 * 60 * 6;

function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));
}
function escTg(s){
  return String(s).replace(/[&<>]/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;"}[c]));
}
function shuffle(arr){
  const a = arr.slice();
  for(let i = a.length - 1; i > 0; i--){
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
function fmtTime(sec){
  const m = Math.floor(sec / 60), s = sec % 60;
  return m + ":" + String(s).padStart(2, "0");
}
function qId(q){ return q.id || ("h_" + q.soru.substring(0, 40).replace(/\s/g, "_")); }

async function fetchJSON(fileName){
  const cacheKey = CACHE_PREFIX + fileName;
  try{
    const cached = JSON.parse(localStorage.getItem(cacheKey) || "null");
    if(cached && Date.now() - cached.t < CACHE_TTL && cached.d) return cached.d;
  }catch(e){}
  const bases = buildBases();
  let lastErr = null;
  for(const base of bases){
    try{
      const res = await fetch(base + fileName + "?v=" + Date.now(), { cache: "no-store" });
      if(!res.ok) throw new Error("HTTP " + res.status);
      const data = await res.json();
      try{ localStorage.setItem(cacheKey, JSON.stringify({ t: Date.now(), d: data })); }catch(e){}
      return data;
    }catch(err){ lastErr = err; }
  }
  throw lastErr || new Error("Bağlantı hatası");
}

/* XP + STREAK */
const LS_XP = "kpss_xp_v1";
function loadXP(){
  try{
    const d = JSON.parse(localStorage.getItem(LS_XP)) || { total: 0, streak: 0, lastDate: null, lessonsDone: [] };
    const today = new Date().toDateString();
    if(d.lastDate !== today){
      const yesterday = new Date(Date.now() - 86400000).toDateString();
      if(d.lastDate !== yesterday){ d.streak = 0; }
    }
    return d;
  }catch(e){ return { total: 0, streak: 0, lastDate: null, lessonsDone: [] }; }
}
function saveXP(d){
  try{ localStorage.setItem(LS_XP, JSON.stringify(d)); }catch(e){}
  renderXP();
}
function addXP(amount){
  const d = loadXP();
  d.total += amount;
  const today = new Date().toDateString();
  if(d.lastDate !== today){ d.streak += 1; d.lastDate = today; }
  saveXP(d);
  showXpPopup(amount);
  return d;
}
function markLessonDone(lessonId){
  const d = loadXP();
  if(!d.lessonsDone.includes(lessonId)){ d.lessonsDone.push(lessonId); }
  saveXP(d);
}
function isLessonDone(id){ return loadXP().lessonsDone.includes(id); }
function renderXP(){
  const d = loadXP();
  const xpEl = $("xpTotal"), stEl = $("streakCount");
  if(xpEl) animateNumber(xpEl, parseInt(xpEl.textContent) || 0, d.total, 700);
  if(stEl) stEl.textContent = d.streak;
}
function showXpPopup(n){
  const p = document.createElement("div");
  p.className = "xp-popup";
  p.innerHTML = `<span>⭐</span> +${n} XP`;
  document.body.appendChild(p);
  setTimeout(() => p.remove(), 1700);
}

    fb.innerHTML = `<div class="fb-title"><span class="ic">✓</span> Doğru!</div><div class="fb-body">${escapeHtml(q.aciklama || "")}</div>`;

    recordAnswer(false);
    btn.classList.remove("dim");
    btn.classList.add("wrong");

    fb.innerHTML = `<div class="fb-title"><span class="ic">✕</span> Yanlış — Konuyu tekrar edelim</div><div class="fb-body">Doğru cevap: <strong>${escapeHtml(q.secenekler[q.dogru])}</strong><br>${escapeHtml(q.aciklama || "")}</div>`;

    // Bu soruyu tekrar kuyruğa ekle
    if(!L.isReview){

    }
    $("lqNextBtn").classList.remove("hidden");
    $("lqNextBtn").focus();
  }
}

function finishLesson(){
  const ders = L.dersler[L.dersIdx];
  const totalQ = L.questions.length;
  const pct = totalQ ? Math.round(L.correct / totalQ * 100) : 0;

  // XP: her doğru +10, ders tamamlama bonusu +20
  let xp = L.correct * 10;
  let bonus = 0;
  if(pct >= 60){ bonus = 20; markLessonDone(ders.id); }
  const totalXp = xp + bonus;

  const xpData = addXP(totalXp);

  $("completeXP").textContent = "+" + totalXp;
  $("completeCorrect").textContent = L.correct;
  $("completeStreak").textContent = xpData.streak;

async function startQuiz(cat, requestedCount){
  S.isFavMode = false;
  S.requestedCount = requestedCount;
  showLoadingOverlay(true);
  let sorular = [];
  try{
    const data = await fetchJSON(cat.dosya);
    if(Array.isArray(data)) sorular = data;
    else if(data && Array.isArray(data.sorular)) sorular = data.sorular;
  }catch(err){ if(FALLBACK_DATA[cat.key]) sorular = FALLBACK_DATA[cat.key].sorular; }
  showLoadingOverlay(false);
  if(sorular.length === 0){ alert("Bu kategoride soru yok."); return; }
  sorular = shuffle(sorular);
  if(requestedCount > 0 && sorular.length > requestedCount) sorular = sorular.slice(0, requestedCount);
  S.list = sorular;
  S.idx = 0; S.correct = 0; S.wrong = []; S.answered = false;
  S.cat = cat;
  $("qTotal").textContent = S.list.length;
  showScreen("quiz");
  startTimer();
  renderQuestion();
}

}

function showLoadingOverlay(show){
  let el = $("loadingOverlay");
  if(!show){ if(el) el.remove(); return; }
  if(el) return;
  el = document.createElement("div"); el.id = "loadingOverlay";
  el.style.cssText = `position:fixed;inset:0;z-index:200;display:grid;place-items:center;background:rgba(6,11,22,.85);backdrop-filter:blur(10px)`;
  el.innerHTML = `<div style="text-align:center"><div style="width:44px;height:44px;border-radius:50%;margin:0 auto 14px;border:3px solid rgba(201,162,39,.18);border-top-color:#e6c455;animation:spin .8s linear infinite"></div><div style="font-size:12px;color:#6b7b93;font-weight:600">Yükleniyor…</div></div>`;
  document.body.appendChild(el);
}

function startTimer(){
  clearInterval(S.timerId);
  S.t0 = Date.now();
  $("qTimer").textContent = "00:00";
  S.timerId = setInterval(() => {
    const sec = Math.floor((Date.now() - S.t0) / 1000);
    $("qTimer").textContent = String(Math.floor(sec / 60)).padStart(2, "0") + ":" + String(sec % 60).padStart(2, "0");
  }, 1000);
}
function stopTimer(){ clearInterval(S.timerId); }

function renderQuestion(){
  const q = S.list[S.idx];
  S.answered = false;
  $("qCounter").textContent = S.idx + 1;
  $("qCat").textContent = S.cat.ad;
  $("qText").textContent = q.soru;
  $("barFill").style.width = (S.idx / S.list.length * 100) + "%";

  const favBtn = $("favToggle");
  if(isFav(q)) favBtn.classList.add("fav-active");
  else favBtn.classList.remove("fav-active");
  $("reportToggle").classList.remove("reported");

  const wrap = $("optionsList");
  wrap.innerHTML = "";
  const indexed = q.secenekler.map((text, i) => ({ text, isCorrect: i === q.dogru }));
  const mixed = shuffle(indexed);
  mixed.forEach((item, i) => {
    const btn = document.createElement("button");
    btn.type = "button"; btn.className = "option";
    btn.dataset.correct = item.isCorrect ? "1" : "0";
    btn.style.animationDelay = (i * 70) + "ms";
    btn.innerHTML = `<span class="letter">${LETTERS[i]}</span><span class="opt-text">${escapeHtml(item.text)}</span><span class="mark"></span>`;
    btn.addEventListener("mousemove", e => {
      const r = btn.getBoundingClientRect();
      btn.style.setProperty("--mx", (e.clientX - r.left) + "px");
      btn.style.setProperty("--my", (e.clientY - r.top) + "px");
    });
    btn.addEventListener("click", () => selectOption(btn, item.isCorrect, q));
    wrap.appendChild(btn);
  });

  const fb = $("feedback"); fb.className = "feedback"; fb.innerHTML = "";
  $("nextBtn").classList.add("hidden");
  $("nextBtn").textContent = (S.idx === S.list.length - 1) ? "Sonuçları Gör →" : "Sonraki →";
}

function selectOption(btn, isCorrect, q){
  if(S.answered) return;
  S.answered = true;
  const all = document.querySelectorAll("#optionsList .option");
  const fb = $("feedback");
  all.forEach(el => {
    el.classList.add("disabled");
    if(el.dataset.correct === "1"){ el.classList.add("correct"); el.querySelector(".mark").textContent = "✓"; }
    else if(el !== btn){ el.classList.add("dim"); }
  });
  if(isCorrect){
    S.correct++;
    recordAnswer(true);
    fb.className = "feedback show correct";
    fb.innerHTML = `<div class="fb-title"><span class="ic">✓</span> Doğru cevap!</div><div class="fb-body">${escapeHtml(q.aciklama || "")}</div>`;
  } else {
    btn.classList.remove("dim"); btn.classList.add("wrong");
    btn.querySelector(".mark").textContent = "✕";
    S.wrong.push({ soru: q.soru, verilen: btn.querySelector(".opt-text").textContent, dogru: q.secenekler[q.dogru] });
    recordAnswer(false);
    fb.className = "feedback show wrong";
    fb.innerHTML = `<div class="fb-title"><span class="ic">✕</span> Yanlış cevap</div><div class="fb-body">Doğru cevap: <strong>${escapeHtml(q.secenekler[q.dogru])}</strong><br>${escapeHtml(q.aciklama || "")}</div>`;
  }
  $("nextBtn").classList.remove("hidden");
  $("nextBtn").focus();
}

function finishQuiz(){
  stopTimer();
  const total = S.list.length;
  const pct = total ? Math.round(S.correct / total * 100) : 0;
  const elapsed = Math.floor((Date.now() - S.t0) / 1000);
  animateNumber($("rPct"), 0, pct, 900);
  animateNumber($("rCorrect"), 0, S.correct, 700);
  animateNumber($("rWrong"), 0, S.wrong.length, 700);
  $("rTime").textContent = fmtTime(elapsed);
  const ring = $("scoreRing");
  let cur = 0;
  const ringAnim = setInterval(() => {
    cur += Math.max(1, Math.ceil(pct / 40));
    if(cur >= pct){ cur = pct; clearInterval(ringAnim); }
    ring.style.background = `conic-gradient(var(--gold) ${cur}%, rgba(255,255,255,.06) ${cur}%)`;
  }, 22);
  let title, msg;
  if(pct >= 90){ title = "Muhteşem! 🏆"; msg = "Bu konuya hâkimsin."; }
  else if(pct >= 75){ title = "Çok iyi! 👏"; msg = "Güçlü performans."; }
  else if(pct >= 50){ title = "İyi gidiyorsun 💪"; msg = "Tekrar ile yükselirsin."; }
  else { title = "Tekrar gerekiyor 📚"; msg = "Yanlışlarını incele."; }
  $("rTitle").textContent = title;
  $("rMsg").textContent = msg;
  const box = $("reviewBox"), list = $("reviewList");
  list.innerHTML = "";
  if(S.wrong.length === 0){ box.classList.add("hidden"); }
  else {
    box.classList.remove("hidden");
    S.wrong.forEach((w, i) => {
      const div = document.createElement("div");
      div.className = "review-item"; div.style.animationDelay = (i * 60) + "ms";
      div.innerHTML = `<div class="rq">${i + 1}. ${escapeHtml(w.soru)}</div><div class="ra"><span>Senin cevabın: <i>${escapeHtml(w.verilen)}</i></span><span>Doğru cevap: <b>${escapeHtml(w.dogru)}</b></span></div>`;
      list.appendChild(div);
    });
  }
  if(pct >= 75) launchConfetti();
  showScreen("result");
}
function animateNumber(el, from, to, dur){
  const t0 = performance.now();
  function tick(t){
    const p = Math.min(1, (t - t0) / dur);
    const eased = 1 - Math.pow(1 - p, 3);
    el.textContent = Math.round(from + (to - from) * eased);
    if(p < 1) requestAnimationFrame(tick);
  }
  requestAnimationFrame(tick);
}
function launchConfetti(){
  const wrap = $("confetti");
  wrap.innerHTML = "";
  const colors = ["#c9a227","#e6c455","#f7e9b8","#34d399","#f87171","#ffffff"];
  for(let i = 0; i < 80; i++){
    const p = document.createElement("i");
    p.style.left = Math.random() * 100 + "%";
    p.style.background = colors[Math.floor(Math.random() * colors.length)];
    p.style.animationDuration = (2.5 + Math.random() * 2) + "s";
    p.style.animationDelay = (Math.random() * 0.5) + "s";
    p.style.opacity = 0.7 + Math.random() * 0.3;
    p.style.transform = `rotate(${Math.random() * 360}deg)`;
    if(Math.random() > 0.5) p.style.borderRadius = "50%";
    wrap.appendChild(p);
    setTimeout(() => p.remove(), 5000);
  }
}

    return;
  }

  const url = `https://api.telegram.org/bot${CONFIG.telegramBotToken}/sendMessage`;
  fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ chat_id: CONFIG.telegramChatId, text: message, parse_mode: "HTML", disable_web_page_preview: true })
  })
  .then(res => res.json())
  .then(data => {
    if(!data.ok) throw new Error(data.description || "Telegram hatası");

    setTimeout(() => repBtn.classList.remove("pop"), 600);

    showToast("❌ Gönderilemedi: " + err.message, true);
  })
  .finally(() => { btn.disabled = false; btn.textContent = orig; });
}
